Renders a list of alternative subdomain suggestions when a user's desired subdomain is already taken, allowing them to quickly select an available alternative.

## Key Components

- **`DomainSuggestions`** — Client component that displays a list of clickable subdomain suggestion buttons.
  - **Props:**
    - `suggestions: string[]` — Array of available subdomain strings to display (component renders `null` if empty).
    - `onSelect: (subdomain: string) => void` — Callback invoked with the chosen subdomain when a suggestion button is clicked.
  - Each suggestion is rendered as an outlined `Button` displaying `{suggestion}.{SAAS_DOMAIN_SUFFIX}` (full domain with suffix appended for clarity).
  - Intended to be rendered into a "slot" of a domain/subdomain input field, typically shown after a signup form detects the requested subdomain is unavailable.

## Usage Example

```typescript
import { useState } from 'react';
import { DomainSuggestions } from './domain-suggestions';

function SignupDomainField() {
  const [suggestions, setSuggestions] = useState<string[]>([]);
  const [subdomain, setSubdomain] = useState('');

  async function handleSubdomainCheck(value: string) {
    const isTaken = await checkSubdomainAvailability(value);
    if (isTaken) {
      setSuggestions(await fetchSubdomainSuggestions(value));
    } else {
      setSuggestions([]);
    }
  }

  return (
    <div>
      <input
        value={subdomain}
        onChange={e => {
          setSubdomain(e.target.value);
          handleSubdomainCheck(e.target.value);
        }}
      />
      <DomainSuggestions
        suggestions={suggestions}
        onSelect={picked => {
          setSubdomain(picked);
          setSuggestions([]);
        }}
      />
    </div>
  );
}
```
