# use-tenant-domain.ts

A React client hook for fetching and updating tenant domain configuration, including the shared auto-provisioning setting that controls whether users with a generic/shared email domain are automatically allowed to join a tenant.

## Key Components

- **`TenantDomainInfo`** — Type describing a tenant's domain config: `domain`, `generic` (whether it's a shared/public domain like gmail.com), and `autoAllow` (auto-provisioning flag).
- **`UpdateSharedAutoProvisionPayload`** — Request payload shape for toggling auto-provisioning.
- **`UpdateSharedAutoProvisionResponse`** — Response shape with optional `success` flag or structured `error` (`code` + `message`).
- **`useTenantDomain()`** — Hook exposing:
  - `fetchTenantDomain()` — Fetches current domain info via `GET api/config/global/domain`, defaulting `autoAllow` to `false` if absent.
  - `updateSharedAutoProvision(enabled)` — Sends `POST api/config/global/domain` with the new `autoAllow` value; returns a structured success/error result instead of throwing for known API error shapes.
  - `isLoading` — `true` while `fetchTenantDomain` is in flight.
  - `isUpdating` — `true` while `updateSharedAutoProvision` is in flight.

## Usage Example

```typescript
'use client';

import { useEffect, useState } from 'react';
import { useTenantDomain, type TenantDomainInfo } from '@/hooks/use-tenant-domain';

export function TenantDomainSettings() {
  const { fetchTenantDomain, updateSharedAutoProvision, isLoading, isUpdating } = useTenantDomain();
  const [domainInfo, setDomainInfo] = useState<TenantDomainInfo | null>(null);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    fetchTenantDomain()
      .then(setDomainInfo)
      .catch((err) => setError(err.message));
  }, [fetchTenantDomain]);

  const handleToggle = async (checked: boolean) => {
    const result = await updateSharedAutoProvision(checked);
    if (result.error) {
      setError(result.error.message);
      return;
    }
    setDomainInfo((prev) => (prev ? { ...prev, autoAllow: checked } : prev));
  };

  if (isLoading || !domainInfo) return <p>Loading...</p>;

  return (
    <div>
      <p>Domain: {domainInfo.domain}</p>
      <label>
        <input
          type="checkbox"
          checked={domainInfo.autoAllow}
          disabled={isUpdating}
          onChange={(e) => handleToggle(e.target.checked)}
        />
        Auto-allow shared domain users
      </label>
      {error && <p role="alert">{error}</p>}
    </div>
  );
}
```
