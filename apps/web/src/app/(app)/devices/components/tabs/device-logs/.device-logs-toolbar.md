# device-logs-toolbar.tsx

A toolbar component for the device logs view that combines search, auto-update toggle, log level filter chips, time range selection, and a manual refresh control. It is rendered outside the log list's Suspense boundary so the search input retains focus during data fetches.

## Key Components

- **`DeviceLogsToolbarProps`** — Interface defining all controlled inputs and callbacks: search text/error, selected log levels, time range (including custom range and date picker bounds), auto-update state, refresh handler/status, and an optional `disabled` flag for when the device is still loading.
- **`DeviceLogsToolbar`** — The main functional component. Renders:
  - An `Input` field for log search (with `SearchIcon` adornment and inline error display)
  - A `CheckboxBlock` for toggling auto-update
  - `DeviceLogLevelChips` for multi-select level filtering
  - `DeviceLogRangeSelect` for choosing predefined or custom date ranges
  - A refresh `Button` with a loading-aware disabled state

## Usage Example

```typescript
import { DeviceLogsToolbar } from './device-logs-toolbar';

function DeviceLogsPanel() {
  const [search, setSearch] = useState('');
  const [selectedLevels, setSelectedLevels] = useState<DeviceLogLevel[]>([]);
  const [range, setRange] = useState<DeviceLogRange>('last_24h');
  const [customRange, setCustomRange] = useState<DateRange | undefined>();
  const [autoUpdate, setAutoUpdate] = useState(true);
  const [isRefreshing, setIsRefreshing] = useState(false);

  return (
    <DeviceLogsToolbar
      search={search}
      onSearchChange={setSearch}
      searchError={null}
      selectedLevels={selectedLevels}
      onToggleLevel={(level) =>
        setSelectedLevels((prev) =>
          prev.includes(level) ? prev.filter((l) => l !== level) : [...prev, level]
        )
      }
      range={range}
      onRangeChange={setRange}
      customRange={customRange}
      onCustomRangeChange={setCustomRange}
      pickerBounds={{ fromDate: new Date(2024, 0, 1), toDate: new Date() }}
      autoUpdate={autoUpdate}
      onAutoUpdateChange={setAutoUpdate}
      onRefresh={() => setIsRefreshing(true)}
      isRefreshing={isRefreshing}
    />
  );
}
```
