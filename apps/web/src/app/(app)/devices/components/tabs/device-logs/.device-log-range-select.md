# device-log-range-select.tsx

A React client component that renders a time-range preset selector for device logs, paired with a conditional date-range picker that appears when the "custom" preset is selected.

## Key Components

- **`DeviceLogRangeSelect`** — Default export function component rendering a `Select` dropdown of predefined log time ranges (`DEVICE_LOG_RANGES`), plus a `DatePicker` for custom date ranges shown only when `range === 'custom'`.
- **`DeviceLogRangeSelectProps`** — Props interface:
  - `range: DeviceLogRange` — current selected preset
  - `onRangeChange: (range: DeviceLogRange) => void` — callback when preset changes
  - `customRange: DateRange | undefined` — selected custom date range
  - `onCustomRangeChange: (range: DateRange | undefined) => void` — callback when custom range changes
  - `pickerBounds: { fromDate: Date; toDate: Date }` — min/max selectable dates for the picker
  - `disabled?: boolean` — disables both controls
  - `triggerClassName?: string` — styling hook for the `Select` trigger
  - `pickerClassName?: string` — styling hook for the `DatePicker`

This component returns a React fragment (no wrapping element), so the caller's layout/row determines how the preset selector and date picker are positioned relative to each other.

## Usage Example

```typescript
import { useState } from 'react';
import { DeviceLogRangeSelect } from './device-log-range-select';
import type { DateRange } from '@flamingo-stack/openframe-frontend-core/components/ui';
import type { DeviceLogRange } from '../../../utils/device-log-time';

function LogsToolbar() {
  const [range, setRange] = useState<DeviceLogRange>('24h');
  const [customRange, setCustomRange] = useState<DateRange | undefined>();

  return (
    <div className="flex items-center gap-2">
      <DeviceLogRangeSelect
        range={range}
        onRangeChange={setRange}
        customRange={customRange}
        onCustomRangeChange={setCustomRange}
        pickerBounds={{ fromDate: new Date('2023-01-01'), toDate: new Date() }}
        triggerClassName="w-40"
        pickerClassName="ml-2"
      />
    </div>
  );
}
```
