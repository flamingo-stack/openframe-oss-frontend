A confirmation dialog shown before permanently deleting a remote session, used from both the session list and the recording playback page.

## Key Components

- **`DeleteSessionDialog`** — React component that wraps the shared `ConfirmDialog` with session-deletion-specific copy, warning the user that the recording and session data are permanently removed.
- **`DeleteSessionDialogProps`** — Props interface:
  - `startedAt`: ISO timestamp string of the session's start time (or `null` if unavailable), formatted and shown in the confirmation message.
  - `open`: controls dialog visibility.
  - `onOpenChange`: callback fired when the dialog's open state changes.
  - `isPending`: disables/shows loading state on the confirm action while the delete request is in flight.
  - `onConfirm`: callback invoked when the user confirms the deletion.

## Usage Example

```typescript
import { useState } from 'react';
import { DeleteSessionDialog } from '@/app/components/sessions/delete-session-dialog';

function SessionRow({ session }: { session: { startedAt: string } }) {
  const [open, setOpen] = useState(false);
  const [isPending, setIsPending] = useState(false);

  async function handleConfirm() {
    setIsPending(true);
    try {
      await deleteSession(session.startedAt);
      setOpen(false);
    } finally {
      setIsPending(false);
    }
  }

  return (
    <>
      <button onClick={() => setOpen(true)}>Delete</button>
      <DeleteSessionDialog
        startedAt={session.startedAt}
        open={open}
        onOpenChange={setOpen}
        isPending={isPending}
        onConfirm={handleConfirm}
      />
    </>
  );
}
```
